cmake_minimum_required(VERSION 3.8)
project(tut3_velocity)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
add_executable(velocity_pub src/velocity_pub.cpp)
ament_target_dependencies(velocity_pub rclcpp geometry_msgs)
add_executable(velocity_sub src/velocity_sub.cpp)
ament_target_dependencies(velocity_sub rclcpp geometry_msgs)
install(TARGETS velocity_pub velocity_sub DESTINATION lib/tut3_velocity)
install(DIRECTORY launch DESTINATION share/tut3_velocity)
ament_package()
