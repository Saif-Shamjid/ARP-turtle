cmake_minimum_required(VERSION 3.8)
project(tut4_sensor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(sensor_pub src/sensor_pub.cpp)
ament_target_dependencies(sensor_pub rclcpp std_msgs)
add_executable(sensor_sub src/sensor_sub.cpp)
ament_target_dependencies(sensor_sub rclcpp std_msgs)
install(TARGETS sensor_pub sensor_sub DESTINATION lib/tut4_sensor)
install(DIRECTORY launch DESTINATION share/tut4_sensor)
ament_package()
