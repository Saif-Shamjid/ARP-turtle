cmake_minimum_required(VERSION 3.8)
project(tut2_counter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(counter_pub src/counter_pub.cpp)
ament_target_dependencies(counter_pub rclcpp std_msgs)
add_executable(counter_sub src/counter_sub.cpp)
ament_target_dependencies(counter_sub rclcpp std_msgs)
install(TARGETS counter_pub counter_sub DESTINATION lib/tut2_counter)
install(DIRECTORY launch DESTINATION share/tut2_counter)
ament_package()
